\chapter{Modelo dinámico}	
\label{cap:modDinamico}

Este capítulo describe el modelo dinámico del sistema SICORRE. En él se detallan 
los escenarios de ejecución correspondientes a las dos primeras iteraciones de 
desarrollo. La figura~\ref{fig:casosDeUso} muestra el diagrama general del sistema 
y sus subsistemas, mientras que la figura~\ref{fig:casosDeUsoDetalle} presenta el 
diagrama detallado con todos los casos de uso identificados. En este documento se 
describen los casos de uso desarrollados durante la iteración 1, correspondiente 
a la gestión de usuarios y equipos multidisciplinarios, y la iteración 2, 
correspondiente al registro y consulta de NNAs y tutores.

\begin{UseCase}{CU-EXT-01}{Buscar dirección por código postal}{
		El sistema consulta el catálogo de SEPOMEX para obtener la información de una dirección a partir de un código postal proporcionado, devolviendo el estado, municipio y colonia correspondientes.
	}
	\UCitem{Versión}{1.0}
	\UCitem{Autor}{Estévez Pérez Alison Beatriz}
	\UCitem{Supervisa}{Pulido Robles Daniel Alfonso}
	\UCitem{Actor}{Sistema (SEPOMEX)}
	\UCitem{Propósito}{\begin{Titemize}
			\Titem Autocompletar los campos de dirección a partir de un código postal, reduciendo errores de captura y agilizando el registro de información geográfica.
		\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
			\Titem Código postal (5 dígitos).
		\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
			\Titem Introducido desde el teclado en el campo de código postal de la interfaz correspondiente al caso de uso que lo invoca.
		\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
			\Titem Estado.
			\Titem Municipio.
			\Titem Colonia(s) disponible(s) para el código postal indicado.
		\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
			\Titem Campos de dirección en la interfaz del caso de uso que lo invoca.
			\Titem Pantalla \IUref{IU-EXT-01}{Auxiliar Direccion}.
		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El usuario ha ingresado un código postal en el campo correspondiente.
			\Titem El catálogo de SEPOMEX se encuentra disponible en el sistema.
		\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
			\Titem Los campos de estado, municipio y colonia quedan autocompletados con la información recuperada del catálogo de SEPOMEX.
		\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
			\Titem {\bf \hypertarget{CU-EXT-01.E1}{E1}}: Código postal no encontrado en el catálogo; el sistema muestra ``No se encontraron resultados para el código postal ingresado'' y limpia los campos de estado, municipio y colonia.
			\Titem {\bf \hypertarget{CU-EXT-01.E2}{E2}}: Código postal con formato inválido (diferente a 5 dígitos numéricos); el sistema muestra ``El código postal debe contener exactamente 5 dígitos'' y no realiza la consulta.
		\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso extendido. Es utilizado por:
		\begin{Titemize}
			\Titem \UCref{CU-04}{Registrar usuario}
			\Titem \UCref{CU-06}{Modificar usuario}
			\Titem \UCref{CU-13}{Registrar NNA}
			\Titem \UCref{CU-18}{Registrar tutor del NNA}
			\Titem \UCref{CU-26}{Registrar actor en materia de derechos}
			\Titem \UCref{CU-28}{Modificar actor en materia de derechos}
			\Titem \UCref{CU-43}{Registrar ubicación del actor}
			\Titem \UCref{CU-49}{Editar actor}
		\end{Titemize}
	}\UCitem{Observaciones}{
		Este caso de uso es invocado automáticamente cada vez que el usuario termina de introducir un código postal válido (5 dígitos) en cualquiera de los casos de uso que lo incluyen. La selección de colonia permanece habilitada para el usuario en caso de que el código postal corresponda a múltiples colonias.
	}
\end{UseCase}
\begin{UCtrayectoria}
	\UCpaso[\UCactor] Introduce un código postal de 5 dígitos en el campo correspondiente de la interfaz activa. \ErrorRef{CU-EXT-01}{E2}{Formato de código postal inválido}
	\UCpaso Valida que el código postal tenga el formato correcto (5 dígitos numéricos).
	\UCpaso \label{CU-EXT-01.consultaCatalogo} Consulta el catálogo de SEPOMEX con el código postal proporcionado. \ErrorRef{CU-EXT-01}{E1}{Código postal no encontrado}
	\UCpaso Recupera el estado, municipio y colonia(s) asociados al código postal.
	\UCpaso Autocompleta los campos de estado y municipio en la interfaz del caso de uso que lo invoca.
	\UCpaso Presenta las colonias disponibles en una lista desplegable para que el usuario seleccione la correspondiente.
	\UCpaso[\UCactor] Selecciona la colonia de la lista desplegable.
	\UCpaso Registra la colonia seleccionada y devuelve el control al caso de uso que invocó este proceso.
\end{UCtrayectoria}

\begin{UseCase}{CU-01}{Iniciar sesión}{
		El usuario del sistema proporciona sus credenciales de acceso para autenticarse e ingresar a los módulos de SICORRE según el rol que tiene asignado.
	}
	\UCitem{Versión}{1.0}
	\UCitem{Autor}{Estévez Pérez Alison Beatriz}
	\UCitem{Supervisa}{Pulido Robles Daniel Alfonso}
	\UCitem{Actor}{\hyperlink{Actor.Usuario}{Usuario (Administrador / Especialista)}}
	\UCitem{Propósito}{\begin{Titemize}
			\Titem Controlar el acceso al sistema para proteger la información confidencial de los NNA.
		\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
			\Titem \hyperlink{Usuarios.correo}{Correo electrónico}.
			\Titem \hyperlink{Usuarios.password_hash}{Contraseña}.
		\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
			\Titem Se introducen desde el teclado en la pantalla de inicio de sesión.
		\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
			\Titem Redirección al panel principal correspondiente a su rol.
			\Titem Mensajes de error en caso de credenciales inválidas.
		\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
			\Titem Pantalla \IUref{IU01}{Iniciar sesión}.
		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El usuario debe estar registrado y activo en el sistema.
		\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
			\Titem Se inicia una sesión activa para el usuario en el sistema.
		\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
			\Titem {\bf \hypertarget{CU-01.E1}{E1}}: Credenciales incorrectas; el sistema muestra ``Correo o contraseña inválidos'' y regresa al paso \ref{CU-01.ingresaCredenciales}.
		\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso primario.
	}\UCitem{Observaciones}{
		Ninguna.
	}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede a la URL del sistema.
	\UCpaso Muestra la pantalla \IUref{IU01}{Inicio de sesión}.
	\UCpaso[\UCactor] \label{CU-01.ingresaCredenciales} Introduce su correo electrónico y contraseña, y presiona \IUbutton{Ingresar}.
	\UCpaso Valida las credenciales ingresadas contra la base de datos. \ErrorRef{CU-01}{E1}{Credenciales incorrectas}
	\UCpaso Verifica el rol asignado al usuario y establece la sesión activa.
	\UCpaso Redirige al panel principal correspondiente a su rol y muestra el mensaje \MSGref{MSG01}{Bienvenida al usuario}.
\end{UCtrayectoria}

\begin{UseCase}{CU-02}{Cerrar sesión}{
		El usuario del sistema selecciona la opción de cierre de sesión disponible en la barra de navegación. El sistema invalida el token de sesión activo y redirige al usuario a la pantalla de inicio de sesión, garantizando que ningún acceso posterior sea posible sin autenticación.
	}
	\UCitem{Versión}{1.0}
	\UCitem{Autor}{Estévez Pérez Alison Beatriz}
	\UCitem{Supervisa}{Pulido Robles Daniel Alfonso}
	\UCitem{Actor}{\hyperlink{Actor.Usuario}{Usuario (Administrador / Especialista)}}
	\UCitem{Propósito}{\begin{Titemize}
			\Titem Invalidar el token de sesión activo para prevenir accesos no autorizados a los datos del sistema una vez que el usuario ha terminado su jornada.
		\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
			\Titem Ninguna (acción iniciada por botón).
		\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
			\Titem Botón \IUbutton{Cerrar sesión} ubicado en la barra de navegación superior.
		\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
			\Titem Pantalla \IUref{IU01}{Inicio de sesión}.
		\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
			\Titem Pantalla \IUref{IU01}{Inicio de sesión}.
		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El usuario debe tener una sesión activa en el sistema.
		\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
			\Titem El token de sesión del usuario es eliminado del servidor.
			\Titem El usuario no puede acceder a ningún módulo protegido sin volver a autenticarse.
		\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
			\Titem Ninguno.
		\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso primario.
	}\UCitem{Observaciones}{
		Ninguna.
	}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Presiona el botón \IUbutton{Cerrar sesión} ubicado en la barra de navegación superior.
	\UCpaso Invalida el token de sesión activo y limpia los datos de sesión del servidor.
	\UCpaso Redirige a la pantalla \IUref{IU01}{Inicio de sesión}.
\end{UCtrayectoria}

\begin{UseCase}{CU-04}{Registrar usuario}{
		El administrador accede al módulo de gestión de usuarios y captura los datos personales, de domicilio y de acceso de un nuevo miembro del equipo multidisciplinario, asignándole un rol específico. El sistema valida que el correo electrónico y la CURP no estén duplicados y genera una contraseña provisional para el nuevo usuario.
	}
	\UCitem{Versión}{1.0}
	\UCitem{Autor}{Estévez Pérez Alison Beatriz}
	\UCitem{Supervisa}{Pulido Robles Daniel Alfonso}
	\UCitem{Actor}{\hyperlink{Actor.Administrador}{Administrador}}
	\UCitem{Propósito}{\begin{Titemize}
			\Titem Otorgar acceso al sistema al personal autorizado de la institución para que pueda operar SICORRE según su rol.
		\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
			\Titem \hyperlink{Usuarios.nombre}{Nombre completo}, \hyperlink{Usuarios.correo}{Correo electrónico}.
			\Titem \hyperlink{Usuarios.curp}{CURP}, \hyperlink{Usuarios.rfc}{RFC}, \hyperlink{Usuarios.fecha_nacimiento}{Fecha de nacimiento}, \hyperlink{Usuarios.sexo}{Sexo}.
			\Titem \hyperlink{Usuarios.tipo_personal}{Tipo de personal}, \hyperlink{Usuarios.activo}{Estado de cuenta}.
			\Titem \hyperlink{Usuarios.rol_id}{Rol del usuario} (Administrador / Especialista).
			\Titem Datos de domicilio: \hyperlink{Direcciones.calle}{Calle}, número exterior, código postal, colonia, tipo de vivienda.
			\Titem \hyperlink{Usuarios.password_hash}{Contraseña provisional}.
		\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
			\Titem Se introducen desde el teclado en la pantalla \IUref{IU04}{Registro de Personal}.
		\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
			\Titem Mensaje \MSGref{MSG06}{Registro exitoso de usuario} al registrar correctamente.
			\Titem El nuevo usuario aparece en el listado de personal activo.
		\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
			\Titem Se muestra en la pantalla \IUref{IU03}{Consulta de Usuarios}.
			\Titem Se almacena en la tabla \hyperlink{Usuarios}{usuarios} y \hyperlink{Direcciones}{direcciones\_usuarios}.
		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El administrador debe haber iniciado sesión (\UCref{CU-01}{Iniciar sesión}).
		\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
			\Titem Se crea un nuevo registro en la tabla \hyperlink{Usuarios}{usuarios} con estado activo.
			\Titem Se registra el domicilio correspondiente en \hyperlink{Direcciones}{direcciones\_usuarios}.
		\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
			\Titem {\bf \hypertarget{CU-04.E1}{E1}}: El correo ingresado ya existe; el sistema muestra el mensaje \MSGref{MSG08}{Correo ya registrado} y regresa al paso \ref{CU-04.ingresaDatos}.
			\Titem {\bf \hypertarget{CU-04.E2}{E2}}: La CURP ingresada ya existe; el sistema muestra el mensaje \MSGref{MSG07}{CURP ya registrada} y regresa al paso \ref{CU-04.ingresaDatos}.
			\Titem {\bf \hypertarget{CU-04.E3}{E3}}: Alguno de los campos obligatorios está vacío; el sistema muestra el mensaje \MSGref{MSG10}{Campos obligatorios incompletos} y regresa al paso \ref{CU-04.ingresaDatos}.
		\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso primario.
	}\UCitem{Observaciones}{
		Ninguna.
	}
\end{UseCase}

\begin{UseCase}{CU-12}{Asignar NNA a equipo multidisciplinario}{
		El administrador del sistema selecciona un NNA registrado y lo asigna a un equipo multidisciplinario activo. El sistema valida que el NNA no esté ya asignado a otro equipo y actualiza el registro correspondiente en la base de datos.

			\Titem \hyperlink{Direcciones.tipo_lugar_id}{Tipo de vivienda} (del catálogo \hyperlink{CatTipoLugar}{cat\_vivienda\_nna}).

		\end{Titemize}

		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El especialista debe haber iniciado sesión en el sistema (\UCref{CU-01}{Iniciar sesión}).

		Caso de uso primario.

	\UCitem{Actor}{\hyperlink{Actor.Especialista}{Usuario (Especialista)}}
	\UCitem{Propósito}{\begin{Titemize}
			\Titem Habilitar de nuevo un programa desactivado cuando vuelva a estar disponible su oferta de atención.
		\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
			\Titem ID del programa.
		\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
			\Titem Se selecciona desde el perfil del actor en la sección de programas deshabilitados.
		\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
			\Titem Mensaje de confirmación de reactivación.
		\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
			\Titem Pantalla de perfil de actor.
		\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
			\Titem El Especialista debe estar autenticado en el sistema.
			\Titem El programa debe estar en estado inactivo.
		\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
			\Titem El campo \texttt{activo} del programa se establece en \texttt{true}.
		\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
			\Titem {\bf \hypertarget{CU-59.E1}{E1}}: El Especialista cancela la confirmación; el sistema descarta la acción sin realizar cambios.
		\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso primario.
	}\UCitem{Observaciones}{
		Ninguna.
	}
\end{UseCase}
\begin{UCtrayectoria}
	\UCpaso[\UCactor] En el listado de programas deshabilitados del perfil del actor, selecciona la opción ``Reactivar'' en el programa deseado.
	\UCpaso Muestra un diálogo de confirmación: ``¿Está seguro de que desea reactivar este programa?''.
	\UCpaso[\UCactor] Confirma la acción presionando \IUbutton{Aceptar}. \ErrorRef{CU-59}{E1}{Acción cancelada}
	\UCpaso Actualiza \texttt{actor\_programa.activo = true} en la base de datos.
	\UCpaso Muestra mensaje ``Programa reactivado exitosamente'' y actualiza el listado.
\end{UCtrayectoria}
